[setup kind]
**install**
curl -Lo ./kind https://kind.sigs.k8s.io/dl/v0.26.0/kind-linux-amd64
chmod +x ./kind
sudo mv ./kind /usr/local/bin/kind
kind --version
**create cluster**
kind create cluster --name <cluter-name>

[kubernetes]
kubectl apply -f <file>
kubectl get pods 
kubectl get svc
kubectl port-forward svc/<service-name> <target_port>:<container_port>
kubectl delete svc/deployment/pod <name>
kubectl create namespace <name>
kubectl config set-context --current --namespace=<namespace_name>
kubectl describe

[helm]
helm create nginx-fullstack

nginx-fullstack/
├── Chart.yaml
├── values.yaml
├── templates/
│   ├── deployment.yaml
│   ├── service.yaml
│   ├── ingress.yaml
│   ├── _helpers.tpl
│   └── ...

#to install new helm
helm install <name> <./chart-folder>

#to upgrade and update configs
helm upgrade <name> <./chart-folder> -f <file>

#to uninstall
helm uninstall <name>

#to view helms
helm list -n <namespace>
